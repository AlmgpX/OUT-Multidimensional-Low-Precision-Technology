cmake_minimum_required(VERSION 3.20)
project(OUT_NATIVE VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This version uses Windows CNG for cryptography. A non-Windows crypto backend is not supplied.")
endif()
find_package(Threads REQUIRED)
set(GUARD_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../Source/OUTGuard/Native)
set(MOTOR_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../Source/OUTMotor/Native)
add_library(out_native STATIC ${GUARD_ROOT}/guard_behavior.cpp ${GUARD_ROOT}/guard_crypto.cpp ${GUARD_ROOT}/guard_server.cpp
  ${MOTOR_ROOT}/OUTMotorNative.cpp ${MOTOR_ROOT}/motor_motion.cpp ${MOTOR_ROOT}/motor_navigation.cpp
  ${MOTOR_ROOT}/motor_view.cpp ${MOTOR_ROOT}/motor_world.cpp ${MOTOR_ROOT}/motor_combat.cpp ${MOTOR_ROOT}/motor_rts.cpp)
target_include_directories(out_native PUBLIC ${GUARD_ROOT} ${MOTOR_ROOT})
target_compile_features(out_native PUBLIC cxx_std_17)
set_target_properties(out_native PROPERTIES OUTPUT_NAME out_native_core)
target_link_libraries(out_native PUBLIC bcrypt Threads::Threads)
if(MSVC)
  target_compile_options(out_native PRIVATE /W4 /O2 /fp:precise /utf-8)
  set_property(TARGET out_native PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_executable(out_native_checks tests/native_checks.cpp)
target_link_libraries(out_native_checks PRIVATE out_native)
if(MSVC)
  target_compile_options(out_native_checks PRIVATE /W4 /O2 /fp:precise /utf-8)
  set_property(TARGET out_native_checks PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
add_test(NAME native_guard_motor COMMAND out_native_checks)
add_library(out_c_header_check OBJECT tests/c_header_check.c)
target_include_directories(out_c_header_check PRIVATE include)
set_property(TARGET out_c_header_check PROPERTY C_STANDARD 11)

add_library(out_native_bridge SHARED src/c_api.cpp)
target_include_directories(out_native_bridge PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(out_native_bridge PRIVATE out_native)
target_compile_features(out_native_bridge PRIVATE cxx_std_17)
set_target_properties(out_native_bridge PROPERTIES OUTPUT_NAME out_native)
if(MSVC)
  target_compile_options(out_native_bridge PRIVATE /W4 /O2 /utf-8)
  set_property(TARGET out_native_bridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
