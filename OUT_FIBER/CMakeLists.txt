cmake_minimum_required(VERSION 3.20)
project(OUT_FIBER VERSION 0.2.0 LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(OUT_FIBER src/main.cpp src/bank.cpp src/world.cpp src/life.cpp src/sandbox.cpp src/content.cpp src/benchmark.cpp tests/selftest.cpp)
target_include_directories(OUT_FIBER PRIVATE include)
target_compile_features(OUT_FIBER PRIVATE cxx_std_17)
target_link_libraries(OUT_FIBER PRIVATE Threads::Threads)
if(WIN32)
  target_sources(OUT_FIBER PRIVATE src/gui_win32.cpp)
  target_link_libraries(OUT_FIBER PRIVATE user32 gdi32 gdiplus)
endif()
if(MSVC)
  target_compile_options(OUT_FIBER PRIVATE /O2 /fp:precise /utf-8 /W4)
  set_property(TARGET OUT_FIBER PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(OUT_FIBER PRIVATE -O3 -fno-fast-math -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME fiber_invariants COMMAND OUT_FIBER --selftest)
